en = look for a new release and install it (asks first)

de = nur nachsehen, nichts installieren
en = only check, install nothing

en = install without asking

en = update exit codes: 0 up to date or updated, 1 error, 3 new release not installed

en = Looking for a new release

[update.notes]

en = Not installed. Later: sshlobby update

# ─── errors ───

[update.err.tool]
de = Für die Update-Prüfung fehlt curl oder wget
en = The update check needs curl or wget

[update.err.http]
de = GitHub meldet einen Fehler: kein Release in %s oder Anfragelimit erreicht
en = GitHub answered with an error: no release in %s, or the request limit is reached

[update.err.net]
de = GitHub nicht erreichbar (%s, Exit-Code %d)
en = GitHub not reachable (%s, exit code %d)

[update.err.answer]
de = Unerwartete Antwort von GitHub (kein Versions-Tag)
en = Unexpected answer from GitHub (no version tag)

[update.err.tmp]
de = Temporärer Ordner lässt sich nicht anlegen
en = Cannot create a temporary folder

[update.err.root]
de = Aktualisieren braucht root: sudo sshlobby update
en = Updating needs root: sudo sshlobby update

[update.err.checkout]
de = Diese Kopie läuft aus einem Quellordner (%s). Dort aktualisieren mit git pull und sudo ./install.sh.
en = This copy runs from a source folder (%s). Update it there with git pull and sudo ./install.sh.

[update.err.location]
de = Unerwarteter Installationsort %s (erwartet: …/sshlobby). Bitte mit install.sh neu installieren.
en = Unexpected install location %s (expected: …/sshlobby). Please reinstall with install.sh.

[update.err.launcher]
de = Kein Starter gefunden, der auf %s zeigt. Bitte mit install.sh neu installieren.
en = No launcher found that points to %s. Please reinstall with install.sh.

[update.err.archive]
de = Archivprüfung fehlgeschlagen: %s. Nichts installiert.
en = Archive check failed: %s. Nothing installed.

[update.err.archive.read]
de = Archiv nicht lesbar
en = cannot read the archive

[update.err.archive.paths]
de = unsichere Einträge (absolute Pfade, .., Links oder mehrere Wurzeln)
en = unsafe entries (absolute paths, .., links or more than one root folder)

[update.err.archive.files]
de = sshlobby.sh, install.sh, lib/ oder lang/ fehlt
en = sshlobby.sh, install.sh, lib/ or lang/ is missing

[update.err.archive.version]
de = Release %s enthält Version %s
en = release %s contains version %s

[update.err.archive.syntax]
de = Syntaxfehler in %s
en = syntax error in %s

[update.err.install]
de = install.sh ist fehlgeschlagen (Exit-Code %d). Die vorherige Version ist wiederhergestellt.
en = install.sh failed (exit code %d). The previous version has been restored.

[update.err.restore]
de = install.sh ist fehlgeschlagen (Exit-Code %d), die vorherige Version ließ sich nicht wiederherstellen. Bitte mit install.sh neu installieren.
en = install.sh failed (exit code %d) and the previous version could not be restored. Please reinstall with install.sh.

[update.err.verify]
de = Nach der Installation meldet sich sshlobby mit „%s“ statt Version %s. Die vorherige Version ist wiederhergestellt.
en = After the installation sshlobby reports "%s" instead of version %s. The previous version has been restored.
